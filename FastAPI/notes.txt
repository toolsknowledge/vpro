step 1.
    create database in mysql

    > mysql -u root -p

    > password : root (my macbook)

    > create database emp_db

    > use emp_db

step 2.
    create requirements.txt

    1) fastapi                  # used to develop Rest API'S
    2) uvicorn[standard]        # this server, used to run api calls
    3) sqlalchemy               # interact with mysql
    4) pymysql
    5) cryptography             # used to connect with mysql database
    6) bcrypt                   # used to  hash the password
    7) pyjwt                    # used to generate token
    8) pydantic[email]          # validate the emails

step 3.
    install libraries

    > python -m venv venv (create virtual env)

    > source venv/bin/activate (activate)

    > pip install -r requirements.txt (install)
            (or)
    > python -m pip install -r requirements.txt
